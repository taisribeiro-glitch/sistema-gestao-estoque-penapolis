import React, { useState } from 'react';
import { PackagePlus, Save, ArrowLeft } from 'lucide-react';

export default function EntradaDoacoes({ setPaginaAtual }) {
  const [formData, setFormData] = useState({
    nome: '',
    categoria: 'Alimentos',
    quantidade: '',
    validade: '',
    doador: '',
    observacao: ''
  });

  const handleSubmit = (e) => {
    e.preventDefault();
    alert(`Doação de "${formData.nome}" registada com sucesso!`);
    if (setPaginaAtual) setPaginaAtual('dashboard');
  };

  return (
    <div className="max-w-3xl mx-auto space-y-6">
      <div className="flex items-center gap-4">
        <button 
          onClick={() => setPaginaAtual && setPaginaAtual('dashboard')}
          className="p-2 text-slate-500 hover:text-slate-800 hover:bg-slate-100 rounded-lg transition-colors"
        >
          <ArrowLeft className="w-5 h-5" />
        </button>
        <div>
          <h2 className="text-2xl font-bold text-slate-900">Registar Nova Doação (Entrada)</h2>
          <p className="text-sm text-slate-600">Insira os dados do item recebido no inventário.</p>
        </div>
      </div>

      <form onSubmit={handleSubmit} className="bg-white p-6 rounded-xl border border-slate-200 shadow-sm space-y-5">
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div className="md:col-span-2">
            <label className="block text-sm font-medium text-slate-700 mb-1">Nome do Item / Produto *</label>
            <input 
              type="text" 
              required
              placeholder="Ex: Cesta Básica, Leite, Sabonete..." 
              value={formData.nome}
              onChange={(e) => setFormData({...formData, nome: e.target.value})}
              className="w-full px-3 py-2 border border-slate-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
            />
          </div>

          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1">Categoria *</label>
            <select 
              value={formData.categoria}
              onChange={(e) => setFormData({...formData, categoria: e.target.value})}
              className="w-full px-3 py-2 border border-slate-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
            >
              <option value="Alimentos">Alimentos</option>
              <option value="Higiene">Higiene Pessoal</option>
              <option value="Limpeza">Material de Limpeza</option>
              <option value="Vestuario">Vestuário</option>
              <option value="Outros">Outros</option>
            </select>
          </div>

          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1">Quantidade *</label>
            <input 
              type="number" 
              min="1"
              required
              placeholder="Ex: 50" 
              value={formData.quantidade}
              onChange={(e) => setFormData({...formData, quantidade: e.target.value})}
              className="w-full px-3 py-2 border border-slate-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
            />
          </div>

          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1">Data de Validade</label>
            <input 
              type="date" 
              value={formData.validade}
              onChange={(e) => setFormData({...formData, validade: e.target.value})}
              className="w-full px-3 py-2 border border-slate-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
            />
          </div>

          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1">Nome do Doador / Parceiro</label>
            <input 
              type="text" 
              placeholder="Ex: Comunidade Local, Empresa Parceira..." 
              value={formData.doador}
              onChange={(e) => setFormData({...formData, doador: e.target.value})}
              className="w-full px-3 py-2 border border-slate-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
            />
          </div>
        </div>

        <div>
          <label className="block text-sm font-medium text-slate-700 mb-1">Observações / Detalhes</label>
          <textarea 
            rows="3"
            placeholder="Informações adicionais sobre o estado dos produtos ou lote..." 
            value={formData.observacao}
            onChange={(e) => setFormData({...formData, observacao: e.target.value})}
            className="w-full px-3 py-2 border border-slate-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
          ></textarea>
        </div>

        <div className="flex justify-end gap-3 pt-4 border-t border-slate-100">
          <button 
            type="button"
            onClick={() => setPaginaAtual && setPaginaAtual('dashboard')}
            className="px-4 py-2 border border-slate-300 text-slate-700 rounded-lg hover:bg-slate-50 transition-colors text-sm font-medium"
          >
            Cancelar
          </button>
          <button 
            type="submit"
            className="bg-emerald-600 hover:bg-emerald-700 text-white px-5 py-2 rounded-lg text-sm font-medium flex items-center gap-2 transition-colors shadow-sm"
          >
            <Save className="w-4 h-4" /> Salvar Entrada
          </button>
        </div>
      </form>
    </div>
  );
}
