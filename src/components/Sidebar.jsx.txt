import React from 'react';
import { 
  LayoutDashboard, 
  PackagePlus, 
  PackageMinus, 
  Boxes, 
  FileText, 
  Users, 
  Settings 
} from 'lucide-react';

export default function Sidebar({ paginaAtual, setPaginaAtual }) {
  const menuItens = [
    { id: 'dashboard', rotulo: 'Painel Geral', icone: LayoutDashboard },
    { id: 'estoque', rotulo: 'Consulta de Estoque', icone: Boxes },
    { id: 'entrada', rotulo: 'Registrar Doação (Entrada)', icone: PackagePlus },
    { id: 'saida', rotulo: 'Dar Baixa (Saída)', icone: PackageMinus },
    { id: 'relatorios', rotulo: 'Relatórios & Extensão', icone: FileText },
    { id: 'usuarios', rotulo: 'Voluntários & Entidades', icone: Users },
  ];

  return (
    <aside className="w-full md:w-64 bg-white border-r border-slate-200 min-h-full p-4 flex flex-col justify-between">
      <div>
        <div className="px-3 py-2 mb-4">
          <p className="text-xs font-semibold text-slate-400 uppercase tracking-wider">
            Navegação Principal
          </p>
        </div>

        <nav className="space-y-1">
          {menuItens.map((item) => {
            const Icone = item.icone;
            const isAtivo = paginaAtual === item.id;

            return (
              <button
                key={item.id}
                onClick={() => setPaginaAtual && setPaginaAtual(item.id)}
                className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm font-medium transition-colors ${
                  isAtivo
                    ? 'bg-emerald-50 text-emerald-700 font-semibold'
                    : 'text-slate-600 hover:bg-slate-50 hover:text-slate-900'
                }`}
              >
                <Icone className={`w-5 h-5 ${isAtivo ? 'text-emerald-600' : 'text-slate-400'}`} />
                <span>{item.rotulo}</span>
              </button>
            );
          })}
        </nav>
      </div>

      {/* Roda-pé da Sidebar */}
      <div className="pt-4 border-t border-slate-200">
        <button
          className="w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm font-medium text-slate-600 hover:bg-slate-50 hover:text-slate-900 transition-colors"
        >
          <Settings className="w-5 h-5 text-slate-400" />
          <span>Configurações</span>
        </button>
      </div>
    </aside>
  );
}
